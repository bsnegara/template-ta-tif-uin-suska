\section{Algoritma}
\begin{algorithm}[htbp]
\caption{Mencari nilai maksimum dalam larik}
\label{alg:max}
\begin{algorithmic}[1]
\Require Larik $A$ dengan $n$ elemen
\Ensure Nilai maksimum pada $A$
\State $maks \gets A[1]$
\For{$i \gets 2$ \textbf{to} $n$}
  \If{$A[i] > maks$}
    \State $maks \gets A[i]$
  \EndIf
\EndFor
\State \Return $maks$
\end{algorithmic}
\end{algorithm}
